(function ($) {
  'use strict';
  if (!$) return;

  $(function () {
    var field = $('#main-reply');
    if (!field.length) return;

    var count = $('#plng');
    if (!count.length) {
      count = $('<small id="plng">символов <b>00</b></small>');
      var wrapper = field.closest('.areafield.required');
      if (wrapper.length) wrapper.append(count);
      else field.after(count);
    }

    function updateCount() {
      var length = String(field.val() || '').length;
      count.find('b').text(length < 10 ? '0' + length : String(length));
    }

    // Preserve the name used by existing quote handlers.
    window.epl3 = updateCount;
    field.off('.resCounterV2').on(
      'input.resCounterV2 change.resCounterV2 keyup.resCounterV2 focus.resCounterV2 mouseout.resCounterV2',
      updateCount
    );
    $(document).off('click.resCounterV2', '.pl-quote, #form-buttons').on(
      'click.resCounterV2', '.pl-quote, #form-buttons',
      function () { setTimeout(updateCount, 100); }
    );
    updateCount();
  });
})(window.jQuery);

<script>
$(function() {
    $('.forum').each(function() {
        const $forum = $(this);

        const tc2Label = $forum
            .find('thead th.tc2')
            .first()
            .text()
            .trim()
            .toLowerCase();

        const tc3Label = $forum
            .find('thead th.tc3')
            .first()
            .text()
            .trim()
            .toLowerCase();

        if (!tc2Label || !tc3Label) return;

        $forum.find('tbody > tr').each(function() {
            const $row = $(this);
            const $tc2 = $row.children('td.tc2');
            const $tc3 = $row.children('td.tc3');

            if (!$tc2.length || !$tc3.length) return;

            const tc2Content = $tc2.html();
            const tc3Content = $tc3.html();

            const $info = $('<td>', {
                class: 'forum_info'
            });

            $('<div>', {
                class: 'tc2'
            })
                .append(
                    $('<em>').text(tc2Label + ': ')
                )
                .append(
                    $('<span>').html(tc2Content)
                )
                .appendTo($info);

            $('<div>', {
                class: 'tc3'
            })
                .append(
                    $('<em>').text(tc3Label + ': ')
                )
                .append(
                    $('<span>').html(tc3Content)
                )
                .appendTo($info);

            $tc2.remove();
            $tc3.remove();

            $info.insertAfter(
                $row.children('td.tcl')
            );
        });
    });
});
</script>

    return str
      /* дефис / короткое тире в начале реплики → длинное тире */
      .replace(/-\s|–\s/gm, '— ')

      /* обычные двойные кавычки */

      .replace(/“([^”]+)”/g, '«$1»')

      /* немецкие типографские кавычки */

  var a = $(this).clone().wrap('<div></div>').parent().html();
  var b = a.replace(/>[^<>]+</gim, replacer);

  if (a != b) {
    $(this).replaceWith(b);
  }
});

<script>
(function () {

    function placeTwinksInMenu() {

        /* Родная кнопка Twink Walker */
        var twinkButton =
            document.getElementById('twk-open');

        if (!twinkButton) {
            return false;
        }


        /* Если уже добавили — второй раз не создаем */
        if (
            document.getElementById('navtwinks')
        ) {
            return true;
        }


        /*
         * На RusFF выход обычно имеет id="navlogout".
         * Если конкретный дизайн его изменил —
         * ниже есть запасной поиск.
         */
        var logout =
            document.getElementById('navlogout');


        if (!logout) {

            var navItems =
                document.querySelectorAll(
                    '#pun-navlinks li'
                );


            for (
                var i = 0;
                i < navItems.length;
                i++
            ) {

                var text =
                    (
                        navItems[i].textContent ||
                        ''
                    )
                        .trim()
                        .toLowerCase();


                if (
                    text === 'exit' ||
                    text === 'выход'
                ) {

                    logout =
                        navItems[i];

                    break;

                }

            }

        }


        /*
         * Если пользователь гость
         * и EXIT отсутствует —
         * ничего не переносим.
         */
        if (!logout) {
            return false;
        }


        /* Создаем новый пункт меню */
        var item =
            document.createElement('li');

        item.id =
            'navtwinks';


        var link =
            document.createElement('a');

        link.href =
            '#';

        link.textContent =
            'ТВИНКИ';


        /*
         * Нажатие в меню просто вызывает
         * уже существующую рабочую кнопку.
         */
        link.addEventListener(
            'click',
            function (event) {

                event.preventDefault();

                twinkButton.click();

            }
        );


        item.appendChild(
            link
        );


        /*
         * Вставляем непосредственно
         * ПЕРЕД EXIT.
         *
         * Админ:
         * ADMIN → TWINKS → EXIT
         *
         * Игрок:
         * MESS. → TWINKS → EXIT
         */
        logout.parentNode.insertBefore(
            item,
            logout
        );


        /*
         * Старую плавающую кнопку
         * справа снизу скрываем.
         */
        twinkButton.style.setProperty(
            'display',
            'none',
            'important'
        );


        return true;

    }



    /*
     * Twink Walker загружается внешним JS,
     * поэтому иногда кнопка появляется
     * чуть позже HTML-низа.
     *
     * Ждем ее автоматически.
     */

    if (
        !placeTwinksInMenu()
    ) {

        var attempts =
            0;


        var timer =
            setInterval(
                function () {

                    attempts++;


                    if (
                        placeTwinksInMenu() ||
                        attempts > 100
                    ) {

                        clearInterval(
                            timer
                        );

                    }

                },
                100
            );

    }

})();
</script>

<script>
(function () {
    function init() {
        var main = document.querySelector('#pun-viewtopic #pun-main');
        var scheduled = false;

        function update() {
            scheduled = false;
            if (!main) return;

            main.querySelectorAll('.post').forEach(function (post) {
                var author = post.querySelector('.post-author');
                var visible = false;

                if (author) {
                    var style = getComputedStyle(author);
                    visible = author.getClientRects().length > 0 &&
                        author.offsetWidth > 0 &&
                        style.visibility !== 'hidden' &&
                        style.display !== 'none';
                }

                if (post.classList.contains('res-center-adjust') !== visible) {
                    post.classList.toggle('res-center-adjust', visible);
                }
            });
        }

        function schedule() {
            if (scheduled) return;
            scheduled = true;
            requestAnimationFrame(update);
        }

        try {
            update();

            if (main) {
                new MutationObserver(schedule).observe(main, {
                    childList: true,
                    subtree: true,
                    attributes: true,
                    attributeFilter: ['class', 'style', 'hidden']
                });

                window.addEventListener('resize', schedule);
                window.addEventListener('load', schedule);
            }
        } finally {
            document.documentElement.classList.remove('res-center-loading');
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
</script>

<script>
(function () {
    'use strict';

    function addTooltip(selector, text) {
        var button = document.querySelector(selector);

        if (!button || button.dataset.resTooltipReady === '1') {
            return !!button;
        }

        button.dataset.resTooltipReady = '1';

        /* убираем системную черную подсказку */
        button.removeAttribute('title');

        var tip = document.createElement('span');
        tip.className = 'res-editor-tooltip';
        tip.textContent = text;

        button.appendChild(tip);

        return true;
    }

    function initTooltips() {
        addTooltip('#button-forum-audio', 'Аудио');
        addTooltip('#button-image2', 'Изображение');
    }

    initTooltips();

    /* обе кнопки могут появиться после загрузки редактора */
    var tries = 0;

    var timer = setInterval(function () {
        tries++;

        initTooltips();

        if (
            (
                document.querySelector('#button-forum-audio .res-editor-tooltip') &&
                document.querySelector('#button-image2 .res-editor-tooltip')
            ) ||
            tries >= 30
        ) {
            clearInterval(timer);
        }
    }, 250);
})();
</script>
