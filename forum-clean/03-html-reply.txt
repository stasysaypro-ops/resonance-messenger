<script>
(function () {
    function addHideProfileButton() {
        const buttonsRow = document.querySelector('#form-buttons tr');
        if (!buttonsRow) return;

        // Если кнопка уже есть, повторно ее не создаем.
        if (document.getElementById('button-hideprofile')) return;

        const td = document.createElement('td');
        td.id = 'button-hideprofile';
        td.title = 'Скрыть профиль';

        const img = document.createElement('img');
        img.src = '/i/blank.gif';
        img.onclick = function () {
            if (typeof bbcode === 'function') {
                bbcode('[hideprofile]', '');
            }
        };

        td.appendChild(img);
        buttonsRow.appendChild(td);
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', addHideProfileButton);
    } else {
        addHideProfileButton();
    }
})();
</script>
