<script>
document.querySelectorAll('.header_banners--wrapper').forEach(function(wrapper) {
	let targetScroll = wrapper.scrollLeft;
	let currentScroll = wrapper.scrollLeft;
	let animationFrame = null;

	function updateFade() {
		const maxScroll = wrapper.scrollWidth - wrapper.clientWidth;
		const scrollLeft = wrapper.scrollLeft;

		wrapper.classList.remove('is-middle', 'is-end', 'no-scroll');

		if (maxScroll <= 1) {
			wrapper.classList.add('no-scroll');
			return;
		}

		if (scrollLeft >= maxScroll - 1) {
			wrapper.classList.add('is-end');
			return;
		}

		if (scrollLeft > 1) wrapper.classList.add('is-middle');
	}

	function animateScroll() {
		currentScroll += (targetScroll - currentScroll) * 0.12;
		wrapper.scrollLeft = currentScroll;
		updateFade();

		if (Math.abs(targetScroll - currentScroll) > 0.5) {
			animationFrame = requestAnimationFrame(animateScroll);
		} else {
			currentScroll = targetScroll;
			wrapper.scrollLeft = targetScroll;
			updateFade();
			animationFrame = null;
		}
	}

	wrapper.addEventListener('wheel', function(e) {
		const maxScroll = wrapper.scrollWidth - wrapper.clientWidth;
		if (maxScroll <= 0) return;

		e.preventDefault();
		targetScroll += e.deltaY;
		targetScroll = Math.max(0, Math.min(targetScroll, maxScroll));

		if (!animationFrame) {
			currentScroll = wrapper.scrollLeft;
			animationFrame = requestAnimationFrame(animateScroll);
		}
	}, { passive: false });

	wrapper.addEventListener('scroll', updateFade, { passive: true });
	window.addEventListener('resize', updateFade);

	updateFade();
});
</script>
